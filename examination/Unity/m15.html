// UGUI：CanvasScaler 的参考分辨率与实际缩放
(double scale, double w, double h) Calc(double sw, double sh, double rw, double rh, double match)
{
    double k = Math.Log(sw / rw, 2) * (1 - match) + Math.Log(sh / rh, 2) * match;
    double scale = Math.Pow(2, k);
    return (scale, sw / scale, sh / scale);
}

double rw = 1080, rh = 1920;
Console.WriteLine($"参考分辨率 {rw}x{rh}");
Console.WriteLine();
Console.WriteLine($"{"屏幕",-14}{"Match",-8}{"缩放",-9}{"逻辑尺寸",-16}结果");
foreach (var (sw, sh, label) in new[] { (1080.0, 1920.0, "1080x1920"), (1080.0, 2340.0, "1080x2340"), (1440.0, 2560.0, "1440x2560"), (2048.0, 1536.0, "2048x1536") })
{
    foreach (var match in new[] { 0.0, 0.5, 1.0 })
    {
        var (s, w, h) = Calc(sw, sh, rw, rh, match);
        string tip = Math.Abs((w / h) - (rw / rh)) > 0.05 ? "比例变了，需锚点适配" : "比例一致";
        Console.WriteLine($"{label,-14}{match,-8}{s,-9:F3}{w,-7:F0}x{h,-7:F0} {tip}");
    }
    Console.WriteLine();
}
Console.WriteLine("Match=0 只按宽度缩放（竖屏游戏常用），Match=1 只按高度缩放（横屏常用）");
Console.WriteLine("Match=0.5 两边折中，是最容易「哪边都不对」的设置，需要配合 Safe Area");

IPointerUpHandler - OnPointerUp - 松开指针时调用（在指针正在点击的游戏对象上调用）（抬起）

IDragHandler - OnDrag - 发生拖动时在拖动对象上调用 （拖拽中）
IEndDragHandler - OnEndDrag - 拖动完成时在拖动对象上调用 （结束拖拽）

IScrollHandler - OnScroll - 当鼠标滚轮滚动时调用
IUpdateSelectedHandler - OnUpdateSelected - 每次勾选时在选定对象上调用
ISelectHandler - OnSelect - 当对象成为选定对象时调用
IDeselectHandler - OnDeselect - 取消选择选定对象时调用
C#
IMoveHandler - OnMove - 发生移动事件（上、下、左、右等）时调用
ISubmitHandler - OnSubmit - 按下 Submit 按钮时调用
ICancelHandler - OnCancel - 按下 Cancel 按钮时调用
