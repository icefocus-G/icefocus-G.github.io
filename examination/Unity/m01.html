// 进阶路线：按依赖关系排出可执行的学习顺序（拓扑排序）
var deps = new Dictionary<string, string[]>
{
    ["C# 补充"] = Array.Empty<string>(),
    ["Unity 入门"] = new[] { "C# 补充" },
    ["Unity 基础"] = new[] { "Unity 入门" },
    ["Unity 核心"] = new[] { "Unity 基础" },
    ["UniTask"] = new[] { "C# 补充" },
    ["UGUI"] = new[] { "Unity 基础" },
    ["TextMeshPro"] = new[] { "UGUI" },
    ["InputSystem"] = new[] { "Unity 基础" },
    ["ScriptableObject"] = new[] { "Unity 核心" },
    ["PlayerPrefs"] = new[] { "Unity 核心" },
    ["Newtonsoft-Json"] = new[] { "C# 补充", "ScriptableObject" },
    ["MessagePack"] = new[] { "Newtonsoft-Json" },
    ["网络开发基础"] = new[] { "MessagePack", "UniTask" },
    ["程序框架"] = new[] { "网络开发基础", "UGUI" },
    ["Addressables"] = new[] { "程序框架" },
    ["YooAsset"] = new[] { "Addressables" },
    ["HybridCLR"] = new[] { "YooAsset" },
    ["性能优化"] = new[] { "Unity 核心", "UGUI" },
    ["背包 10w"] = new[] { "性能优化" },
    ["iOS 打包"] = new[] { "性能优化" },
    ["Android 打包"] = new[] { "性能优化" }
};

var done = new List<string>();
var left = new Dictionary<string, string[]>(deps);
int round = 0;
while (left.Count > 0)
{
    var ready = left.Where(kv => kv.Value.All(done.Contains)).Select(kv => kv.Key).OrderBy(x => x).ToList();
    if (ready.Count == 0) { Console.WriteLine("存在循环依赖，无法排序"); break; }
    round++;
    Console.WriteLine($"第 {round} 批：{string.Join(" · ", ready)}");
    foreach (var r in ready) { done.Add(r); left.Remove(r); }
}
Console.WriteLine();
Console.WriteLine($"共 {done.Count} 个模块，{round} 批学完");

进阶六阶段与验收标准

原稿正文只有一张大纲图，这里把「学到什么程度算过」写成可自检的标准。每一阶段都以「能独立做出来」为终点，做不出来就不要往下走，否则后面全是夹生饭。

① 语言补齐：C# 与异步

目标是把 C# 当成工具而不是语法题。必须能徒手写出事件中心（订阅/退订/派发）、用 async/await 串起「加载 → 初始化 → 进场景」的流程、说清闭包为什么会捕获到循环变量的最后一个值。

  • 代表坑：事件忘记退订导致对象无法释放；LINQ 延迟执行被多次枚举；在 Update 里 new 对象制造 GC。
  • 验收：给定一个「加载配置 → 拉取玩家数据 → 打开主界面」的需求，能写出不阻塞主线程、异常可捕获、可取消的实现。

② 引擎打底：生命周期与坐标

Unity 的所有诡异 bug，八成出在「我以为它已经准备好了」。必须能在一张纸上画出 Awake → OnEnable → Start → FixedUpdate → Update → LateUpdate → OnDisable → OnDestroy 的顺序，并解释为什么在 Awake 里拿不到别的物体的引用。

  • 代表坑：脚本执行顺序未设置；Update 里做物理；世界坐标与本地坐标混用。
  • 验收：能说清「角色移动抖动」的三种成因与对应改法。

③ 表现与输入

UI 与输入是玩家唯一直接接触的部分。目标是同一套界面在 16:9 / 4:3 / 19.5:9 下都不错位，输入不写两套（新老输入系统不混用）。

  • 代表坑：CanvasScaler 的 Match 随便设；图集没规划导致 Draw Call 爆炸；TMP 字体图集溢出后文字变方块。
  • 验收：换三种分辨率截图对比，无一个元素溢出安全区。

④ 数据与网络

这一层决定项目能不能长大。存档要能加字段不炸老档，协议要能加字段不炸老客户端。

  • 代表坑：用 JsonUtility 存字典；协议没有版本号；把 TCP 当消息流用，不处理粘包。
  • 验收：给出一份带版本号的存档结构 + 一套含长度前缀、心跳、重连的协议设计。

⑤ 框架与热更

从「能跑」到「能长期维护」。要求能独立出资源热更包，并把新增业务逻辑以 DLL 形式下发，且新 DLL 里引用的类在主包中一定存在。

  • 代表坑：热更资源挂了主包没有的脚本；AOT 泛型实例缺失导致运行时崩；热更代码与主包代码版本不匹配。
  • 验收：只发一个资源包 + 一个逻辑包，客户端重启后表现正确。

⑥ 优化与上线

优化不是「把设置调到最高」，而是有目标值、有度量、有回退方案。上线则是把前面所有坑一次性验收。

  • 代表坑：Overdraw 过高；GC 峰值卡顿；Target API 不达标被拒；iOS 隐私清单缺失。
  • 验收：给出 Draw Call / 内存 / 帧率三张曲线，并说明每项优化前后的差值。
!
顺序不能跳。①② 是地基，跳过会在 ③ 之后反复返工；⑤⑥ 是放大器，地基不稳时优化只会让 bug 更难定位。
