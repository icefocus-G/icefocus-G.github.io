// 视频播放器状态机：只允许合法的状态转移
var allowed = new Dictionary<string, string[]>
{
    ["Idle"]     = new[] { "Prepared" },
    ["Prepared"] = new[] { "Playing", "Idle" },
    ["Playing"]  = new[] { "Paused", "Stopped" },
    ["Paused"]   = new[] { "Playing", "Stopped" },
    ["Stopped"]  = new[] { "Prepared", "Idle" }
};

string state = "Idle";
bool Can(string to) => allowed.TryGetValue(state, out var tos) && tos.Contains(to);

void Go(string to)
{
    if (Can(to)) { Console.WriteLine($"  {state,-9} -> {to,-9}  允许"); state = to; }
    else Console.WriteLine($"  {state,-9} -> {to,-9}  拒绝（非法转移）");
}

Console.WriteLine("正常流程：Prepare -> Play -> Pause -> Play -> Stop");
Go("Prepared"); Go("Playing"); Go("Paused"); Go("Playing"); Go("Stopped");
Console.WriteLine();
Console.WriteLine("常见错误流程：Idle 直接 Play、Stopped 直接 Play");
state = "Idle"; Go("Playing");
state = "Stopped"; Go("Playing");
Console.WriteLine();
Console.WriteLine($"最终状态：{state}");

VideoPlayer 能力边界

Unity 自带的 VideoPlayer 本质是把平台解码器包了一层：Windows/macOS 走系统解码，Android 走 ExoPlayer，iOS 走 AVFoundation。所以「同一个视频在编辑器和真机表现不同」是常态，不是 bug。

能力是否支持说明
常见封装mp4(H.264) / webm(VP8)H.265、AV1 在部分安卓机型直接黑屏，发布前必须真机验证
透明视频仅部分平台需要带 alpha 的编码，安卓常需转成 Chroma Key 方案
音频输出AudioSource 模式 / 直出要混音、要 3D 衰减就必须挂 AudioSource
拖动定位支持 seek在没有关键帧的位置拖动会花屏或卡顿，需 keyframe 密集编码
多视频同屏受限移动端同时解码路数有限，超过 2~3 路会掉帧

Render Mode 三种输出方式

  • Camera Far Plane / Near Plane：直接贴到相机平面，最省性能，适合全屏过场动画；缺点是无法被 UI 遮挡。
  • Render Texture：渲染到 RT 再挂到 RawImage 或材质上，最灵活，可做 UI 内嵌播放器；代价是多一次显存拷贝。
  • Material Override：直接替换某个材质的贴图，适合场景内电视屏幕、广告牌。

一次可靠的播放流程

顺序错了就会出现「黑屏两秒」「有画面没声音」「循环播放每次都闪一下」。正确顺序是：先 Prepare，等 prepareCompleted，再 Play。

using UnityEngine;
using UnityEngine.Video;

public class SimplePlayer : MonoBehaviour
{
    public VideoPlayer vp;
    public string url;

    void Awake()
    {
        vp.source = VideoSource.Url;
        vp.url = url;
        vp.playOnAwake = false;
        vp.isLooping = true;
        vp.renderMode = VideoRenderMode.RenderTexture;
        vp.audioOutputMode = VideoAudioOutputMode.AudioSource;
        vp.EnableAudioTrack(0, true);
        vp.SetTargetAudioSource(0, GetComponent<AudioSource>());

        vp.prepareCompleted += OnPrepared;
        vp.errorReceived += (p, msg) => Debug.LogError("视频错误: " + msg);
        vp.Prepare();          // 先准备，不要直接 Play
    }

    void OnPrepared(VideoPlayer p)
    {
        p.Play();
    }

    void OnDestroy()
    {
        vp.prepareCompleted -= OnPrepared;
        vp.Stop();
    }
}

五个高频坑

  • 切场景后视频不释放：VideoPlayer 持有解码器，务必在 OnDestroy 里 Stop 并置空 targetTexture，否则来回切几次就 OOM。
  • seek 之后立刻读 frame：seek 是异步的，要等 seekCompleted 回调，否则拿到的是旧帧。
  • 用 RenderTexture 忘了 Release：手动 new 的 RT 要自己 Release，否则显存只增不减。
  • 移动端 URL 用 http：iOS 默认 ATS 禁止明文 http，必须 https 或在 Info.plist 里例外。
  • 把视频当 UI 背景还开了 Mask：Mask 会产生额外 Draw Call 与 RT 切换，能不用就不用。
