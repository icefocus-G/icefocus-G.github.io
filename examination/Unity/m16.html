// TextMeshPro：SDF 字体为什么放大不糊
double Sdf(double x, double y, double r) => r - Math.Sqrt(x * x + y * y);   // 圆的符号距离场

Console.WriteLine("SDF 值 = 到边界的带符号距离（内部正、外部负）");
foreach (var (x, label) in new[] { (0.0, "圆心"), (0.5, "半径一半"), (0.95, "接近边缘"), (1.0, "边缘上"), (1.2, "外部") })
{
    Console.WriteLine($"  (x={x:F2}, y=0)  SDF={Sdf(x, 0, 1):F3}  {label}");
}

Console.WriteLine();
Console.WriteLine("边缘 = SDF 变号的位置，用 smoothstep 抗锯齿：");
for (int i = 0; i <= 10; i++)
{
    double sdf = -0.3 + i * 0.06;
    double alpha = Math.Clamp((sdf + 0.05) / 0.10, 0, 1);   // 0.1 宽的过渡带
    int bar = (int)Math.Round(alpha * 20);
    Console.WriteLine($"  SDF={sdf,6:F2}  alpha={alpha:F2}  {new string('#', bar)}{new string('.', 20 - bar)}");
}

08：3D文本相关

2D Object文字还原

当成一个 3D 组件使用即可

C#
1.组件类型不同
  3D:TextMeshPro
  UI:TextMeshProUGUI
2.使用场景不同
  3D:用于3D世界空间显示文本
  UI:用于UI系统显示，放在UI画布下面
3.渲染方式
  3D:直接渲染物体网格
  UI:通过UGUI Canvas系统渲染
4.接收事件方式
  3D:需要碰撞体，射线检测
  UI:依托UI系统事件，可以使用EventTrigger
总结
如果需要3D世界内显示文字，选择3D版本TextMeshPro，挂载到3D物体上
如果要在UI系统里面使用，选择 TextMeshProUGUI，放在UI画布下使用

10：TMP_TextUtilities类

继承IPointerClickHandler 接口

C#
using System.Collections;
using System.Collections.Generic;
using TMPro;
using UnityEngine;
using UnityEngine.EventSystems;

public class Lesson12 : MonoBehaviour, IPointerClickHandler
{
    public TextMeshProUGUI tmpUIText;

    public void OnPointerClick(PointerEventData eventData)
    {
        print("123");
        //查找点击位置命中的link链接索引
        int linkIndex = TMP_TextUtilities.FindIntersectingLink(tmpUIText, eventData.position, null);
        //返回值为‑1 代表没有点击到链接
        if(linkIndex != -1)
        {
            //获取链接对应的显示文本信息
            print(tmpUIText.textInfo.linkInfo[linkIndex].GetLinkText());
            //获取<link=?>标签中设置的id携带信息
            print(tmpUIText.textInfo.linkInfo[linkIndex].GetLinkID());
        }

        //查找点击位置命中的字符索引，true = 只识别可见字符
        int charIndex = TMP_TextUtilities.FindIntersectingCharacter(tmpUIText, eventData.position, null, true);
        if(charIndex != -1)
        {
            print(tmpUIText.textInfo.characterInfo[charIndex].character);
        }
    }

 
 
}
C#
这是 TextMeshPro 提供的工具静态类
用来根据屏幕坐标获取点击位置对应的文本信息
做点击文本交互的时候就需要使用这套工具
C#
