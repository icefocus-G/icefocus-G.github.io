// Addressables：引用计数与卸载时机
var refCount = new Dictionary<string, int>();
var loaded = new HashSet<string>();

void Load(string key)
{
    if (!refCount.ContainsKey(key)) { refCount[key] = 0; loaded.Add(key); }
    refCount[key]++;
    Console.WriteLine($"  Load   {key,-16} 引用数 -> {refCount[key]}");
}

void Release(string key)
{
    if (!refCount.ContainsKey(key) || refCount[key] == 0) { Console.WriteLine($"  Release {key,-15} 警告：未加载"); return; }
    refCount[key]--;
    Console.WriteLine($"  Release {key,-15} 引用数 -> {refCount[key]}");
    if (refCount[key] == 0) { loaded.Remove(key); Console.WriteLine($"          -> 引用归零，真正卸载 {key}"); }
}

第 2 章：Addressables 概述与导入

第 1 节：概述

一、Addressables 是什么 可寻址资源管理系统，Unity 2018.2 起推出、用于替代 AssetBundle 的高阶资源管理系统；本质是基于 AssetBundle 架构做的高阶流程（自动化封装）。补充说明 AssetBundle 目前仍是主流但将逐渐被淘汰。

二、Addressables 用于做什么 与 AssetBundle 作用一致，核心三点：管理资源、热更新、减小包的体积。

三、有什么优缺点

  • 优点（5 条）：自动化管理 AB 包打包/发布/加载、更方便本地与远程加载、自动处理资源关联性、内存管理更方便、迭代更方便。
  • 缺点：原 PDF 未明确列出，仅从"学习前提"可归纳出隐性门槛——需先掌握委托事件、AssetBundle 等前置知识，学习门槛相对较高（已在文档中如实标注，未凭空编造）。

     //Action<AsyncOperationHandle<T>> callBack  一个有参无返回值的委托
        //由于存在同名 不同类型资源的区分加载

        AsyncOperationHandle<T> handle;

        if (resDic.ContainsKey(keyName))
        {
            //获取异步加载返回的操作内容，把字典中存储的 异步加载返回值取出来并 强转
            handle = (AsyncOperationHandle<T>)resDic[keyName];
            //判断 这个异步加载是否结束
            if(handle.IsDone)
            {
                //如果成功 就不需要异步了 直接相当于同步调用了 这个委托函数 传入对应的返回值
                callBack(handle);//执行委托 并传入定义委托时候的方法。
    //C#给委托调用做了语法糖：callBack(handle)在编译时会被翻译成 callBack.Invoke(handle)
            }
            //还没有加载完成
            else
            {
                //如果这个时候 还没有异步加载完成 那么我们只需要 告诉它 完成时做什么就行了
                handle.Completed += (obj) => {
                    if (obj.Status == AsyncOperationStatus.Succeeded)
                        callBack(obj);
                };
            }
            return;
        }

        //直接进行异步加载 并且记录
        handle = Addressables.LoadAssetAsync<T>(name);
        handle.Completed += (obj)=> {
        if (obj.Status == AsyncOperationStatus.Succeeded)
                callBack(obj);
        else
        {
                Debug.LogWarning(keyName + "资源加载失败");
                if(resDic.ContainsKey(keyName))
                    resDic.Remove(keyName);
        }
        };
        resDic.Add(keyName, handle);
}

using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;
public class AddressablesMgr
{
    private static AddressablesMgr instance = new AddressablesMgr();
    public static AddressablesMgr Instance => instance;
    //缓存加载出来的资源句柄
    public Dictionary<string, IEnumerator> resDic = new Dictionary<string, IEnumerator>();
    private AddressablesMgr() { }
    //加载单个资源
    public void LoadAssetAsync<T>(string name, Action<AsyncOperationHandle<T>> callBack)
    {
        //处理不同类型同名资源
        //通过名称加类型作为 key
        string keyName = name + "_" + typeof(T).Name;
        AsyncOperationHandle<T> handle;
        //判断是否有缓存
        if (resDic.ContainsKey(keyName))
        {
            //取出缓存的句柄
            handle = (AsyncOperationHandle<T>)resDic[keyName];
            //判断是否加载完成
            if(handle.IsDone)
            {
                //加载成功 不需要等待 直接相当于同步调用 执行回调 使用缓存的句柄
                callBack(handle);
            }
            //没有完成
            else
            {
                //加载中状态 没有加载完成 我们只需要 等待 不需要做其他操作
                handle.Completed += (obj) => {
                    if (obj.Status == AsyncOperationStatus.Succeeded)
                        callBack(obj);
                };
            }
            return;
        }
        
        //没有缓存
        //直接发起加载 请求
        handle = Addressables.LoadAssetAsync<T>(name);
        handle.Completed += (obj)=> {
            if (obj.Status == AsyncOperationStatus.Succeeded)
                callBack(obj);
            else
            {
                Debug.LogWarning(keyName + "资源加载失败");
                if(resDic.ContainsKey(keyName))
                    resDic.Remove(keyName);
            }
        };
        resDic.Add(keyName, handle);
    }
    //释放单个资源 
    public void Release<T>(string name)
    {
        //处理不同类型同名资源
        //通过名称加类型作为 key
        string keyName = name + "_" + typeof(T).Name;
        if(resDic.ContainsKey(keyName))
        {
            //取出句柄 释放资源 删除字典数据
            AsyncOperationHandle<T> handle = (AsyncOperationHandle<T>)resDic[keyName];
            Addressables.Release(handle);
            resDic.Remove(keyName);
        }
    }
    //加载多个资源 回调 返回每个资源
    public void LoadAssetAsync<T>(Addressables.MergeMode mode, Action<T> callBack, params string[] keys)
    {
        //1.拼接一个keyName 之后存到字典
        List<string> list = new List<string>(keys);
        string keyName = "";
        foreach (string key in list)
            keyName += key + "_";
        keyName += typeof(T).Name;
        //2.判断是否已经有缓存 
        AsyncOperationHandle<IList<T>> handle;
        if (resDic.ContainsKey(keyName))
        {
            handle = (AsyncOperationHandle<IList<T>>)resDic[keyName];
            //判断是否完成
            if(handle.IsDone)
            {
                foreach (T item in handle.Result)
                    callBack(item);
            }
            else
            {
                handle.Completed += (obj) =>
                {
                    //加载成功遍历结果回调
                    if(obj.Status == AsyncOperationStatus.Succeeded)
                    {
                        foreach (T item in handle.Result)
                            callBack(item);
                    }
                };
            }
            return;
        }
        //发起加载
        handle = Addressables.LoadAssetsAsync<T>(list, callBack, mode);
        handle.Completed += (obj) =>
        {
            if(obj.Status == AsyncOperationStatus.Failed)
            {
                Debug.LogError("资源加载失败" + keyName);
                if (resDic.ContainsKey(keyName))
                    resDic.Remove(keyName);
            }
        };
        resDic.Add(keyName, handle);
    }

using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;

public class AddressablesMgr
{
    private static AddressablesMgr instance = new AddressablesMgr();
    public static AddressablesMgr Instance => instance;
    //有一个容器 帮助我们存储 异步加载的返回值
    public Dictionary<string, AsyncOperationHandle> resDic = new Dictionary<string, AsyncOperationHandle>();
    private AddressablesMgr() { }

    //异步加载资源的方法
    public void LoadAssetAsync<T>(string name, Action<AsyncOperationHandle<T>> callBack)
    {
        //由于存在同名 不同类型资源的区分加载
        //所以我们通过名字和类型拼接作为 key
        string keyName = name + "_" + typeof(T).Name;
        AsyncOperationHandle<T> handle;
        //如果已经加载过该资源
        if (resDic.ContainsKey(keyName))
        {
            //获取异步加载返回的操作内容
            handle = resDic[keyName].Convert<T>();
            //判断 这个异步加载是否结束
            if(handle.IsDone)
            {
                //如果成功 就不需要异步了 直接相当于同步调用了 这个委托函数 传入对应的返回值
                callBack(handle);
            }
            //还没有加载完成
            else
            {
                //如果这个时候 还没有异步加载完成 那么我们只需要 告诉它 完成时做什么就行了
                handle.Completed += (obj) => {
                    if (obj.Status == AsyncOperationStatus.Succeeded)
                        callBack(obj);
                };
            }
            return;
        }
        
        //如果没有加载过该资源
        //直接进行异步加载 并且记录
        handle = Addressables.LoadAssetAsync<T>(name);
        handle.Completed += (obj)=> {
            if (obj.Status == AsyncOperationStatus.Succeeded)
                callBack(obj);
            else
            {
                Debug.LogWarning(keyName + "资源加载失败");
                if(resDic.ContainsKey(keyName))
                    resDic.Remove(keyName);
            }
        };
        resDic.Add(keyName, handle);
    }

    //释放资源的方法 
    public void Release<T>(string name)
    {
        //由于存在同名 不同类型资源的区分加载
        //所以我们通过名字和类型拼接作为 key
        string keyName = name + "_" + typeof(T).Name;
        if(resDic.ContainsKey(keyName))
        {
            //取出对象 移除资源 并且从字典里面移除
            AsyncOperationHandle<T> handle = resDic[keyName].Convert<T>();
            Addressables.Release(handle);
            resDic.Remove(keyName);
        }
    }

    //异步加载多个资源 或者 加载指定资源
    public void LoadAssetAsync<T>(Addressables.MergeMode mode, Action<T> callBack, params string[] keys)
    {
        //1.构建一个keyName 之后用于存入到字典中
        List<string> list = new List<string>(keys);
        string keyName = "";
        foreach (string key in list)
            keyName += key + "_";
        keyName += typeof(T).Name;

（🔥 必看）05 引用计数规则 知识点（重要）

C#
当我们通过加载使用可寻址资源时
Addressables会在内部帮助我们进行引用计数
使用资源时，引用计数+1
释放资源时，引用计数-1
当可寻址资源的引用为0时，就可以卸载它了

为了避免内存泄露（不需要使用的内容残留在内存中）
我们要保证加载资源和卸载资源是配对使用的

注意：释放的资源不一定立即从内存中卸载
在卸载资源所属的AB包之前，不会释放资源使用的内存
(比如自己所在的AB包 被别人使用时，这时AB包不会被卸载，所以自己还在内存中)
我们可以使用Resources.UnloadUnusedAssets卸载资源（建议在切换场景时调用）

AB包也有自己的引用计数（Addressables把它也视为可寻址资源）
从AB包中加载资源时，引用计数+1
从AB包中卸载资源时，引用计数-1
当AB包引用计数为0时，意味着不再使用了，这时会从内存中卸载

总结：Addressables内部会通过引用计数帮助我们管理内存
我们只需要保证 加载和卸载资源配对使用即可

    public void LoadAssetAsync<T>(string name, Action<AsyncOperationHandle<T>> callBack)
    {
        //由于存在同名 不同类型资源的区分加载
        //所以我们通过名字和类型拼接作为 key
        string keyName = name + "_" + typeof(T).Name;
        AsyncOperationHandle<T> handle;
        //如果已经加载过该资源
        if (resDic.ContainsKey(keyName))
        {
            //获取异步加载返回的操作内容
            handle = resDic[keyName].handle.Convert<T>();
            //要使用资源了 那么引用计数+1
            resDic[keyName].count += 1;
            //判断 这个异步加载是否结束
            if(handle.IsDone)
            {
                //如果成功 就不需要异步了 直接相当于同步调用了 这个委托函数 传入对应的返回值
                callBack(handle);
            }
            //还没有加载完成
            else
            {
                //如果这个时候 还没有异步加载完成 那么我们只需要 告诉它 完成时做什么就行了
                handle.Completed += (obj) => {
                    if (obj.Status == AsyncOperationStatus.Succeeded)
                        callBack(obj);
                };
            }
            return;
        }
        
        //如果没有加载过该资源
        //直接进行异步加载 并且记录

        handle.Completed += (obj)=> {
            if (obj.Status == AsyncOperationStatus.Succeeded)
                callBack(obj);

（🔥 必看）第 3 节：常见问题汇总

C#
1.AB包太多(太小)的问题
  1-1:每个包都会有内存开销，如果加载过多的包可能会带来更多的内存消耗
  1-2:并发下载的问题，如果包小而多，以为着下载包时可能需要更多的时间
  1-3:目录文件会因为过多的包而变大，因为它要记录更多的包信息
  1-4:复用资源的可能性更大，比如多个包使用同样一个资源，但是该资源不是寻址资源，那么在每个包中都会有该资源

2.AB包太少(太大)的问题
  1-1:过大的包如果下载失败，下次会重新下载，因为使用UnityWebRequest下载包时不会恢复失败的下载
      比如100MB的包，下了50MB，玩家中断下载了，下次又得重新下
  1-2:能单独加载，但是不能单独卸载，更大的包意味着包中有更多资源，比如加载了1个大包中100个资源
      但是现在用完了99个，还剩一个再用，即使99都卸载了，但是由于引用计数这个大包也不会卸载
      就会造成内存的浪费

所以没有最好的策略，只有根据自己的项目需求合理安排分组打包
要根据资源的使用情况来合理设置资源分组，在分组时权衡好各方面的问题
C#
AB包的压缩方式：不压缩、LZ4、LZMA
一般情况LZ4用于本地资源，LZMA用于远端资源
主要原因是LZMA的压缩内容更小，更节约下载时间和流量
注意：压缩不会影响加载内存的大小，只会影响包体大小，下载时间等

但是也要根据实际情况
比如：
1.不压缩：包体并不大的单机游戏，使用不压缩最好，没有包体大小的压力，加载也是最快的，因为不用解压
2.LZ4：它是基于块的压缩，所以提供了加载文件的能力，加载资源时不用全加载AB包，只加载使用的内容，相对LZMA来说更节约内存
3.LZMA：不建议用它在本地内容中，因为它虽然包最小，但是加载最慢，用它只是为了节约下载时间和极限压缩包体大小

综合来说，也没有最优的方式，还是要根据实际情况来选择
个人认为LZ4压缩方式，是相对比较优秀的一种方式

  但是我们就只能使用老方法加载同名
  个人建议取消，因为一般我们不会通过Addressables去加载非寻址资源

C#
using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;

/// <summary>
/// 可寻址资源管理器（单例）
/// 支持引用计数、单个/批量异步加载、资源释放及全局清理
/// </summary>
public class AddressablesMgr
{
    // ---------- 单例实现 ----------
    private static readonly Lazy<AddressablesMgr> _instance = new Lazy<AddressablesMgr>(() => new AddressablesMgr());
    public static AddressablesMgr Instance => _instance.Value;

    // ---------- 私有成员 ----------
    // 使用复合 Key（资源名 + 类型名）避免字符串拼接带来的开销和歧义
    private readonly Dictionary<ResourceKey, AddressablesInfo> _resDic = new Dictionary<ResourceKey, AddressablesInfo>();

    private AddressablesMgr() { }

    // ---------- 公共 API ----------

    /// <summary>
    /// 异步加载单个资源（支持引用计数）
    /// </summary>
    /// <typeparam name="T">资源类型</typeparam>
    /// <param name="name">资源地址</param>
    /// <param name="callback">加载完成回调（成功时调用）</param>
    public void LoadAssetAsync<T>(string name, Action<AsyncOperationHandle<T>> callback)
    {
        var key = new ResourceKey(name, typeof(T));
        AsyncOperationHandle<T> handle;

        if (_resDic.TryGetValue(key, out var info))
        {
            // 已加载：增加引用计数
            info.count++;
            handle = info.handle.Convert<T>();

            if (handle.IsDone)
            {
                // 若已加载完成，直接回调
                callback?.Invoke(handle);
            }
            else
            {
                // 若未完成，注册完成回调
                handle.Completed += op =>
                {
                    if (op.Status == AsyncOperationStatus.Succeeded)
                        callback?.Invoke(op);
                    else
                        Debug.LogWarning($"[AddressablesMgr] 资源 {key} 加载失败（已完成但状态异常）");
                };
            }
            return;
        }

        // 首次加载
        handle = Addressables.LoadAssetAsync<T>(name);
        handle.Completed += op =>
        {
            if (op.Status == AsyncOperationStatus.Succeeded)
            {
                callback?.Invoke(op);
            }
            else
            {
                Debug.LogError($"[AddressablesMgr] 资源 {key} 加载失败");
                // 加载失败则移除记录（避免残留）
                if (_resDic.ContainsKey(key))
                    _resDic.Remove(key);
            }
        };

        _resDic.Add(key, new AddressablesInfo(handle));
    }

    /// <summary>
    /// 释放单个资源（引用计数减1，计数为0时真正释放）
    /// </summary>
    /// <typeparam name="T">资源类型</typeparam>
    /// <param name="name">资源地址</param>
    public void Release<T>(string name)
    {
        var key = new ResourceKey(name, typeof(T));
        if (!_resDic.TryGetValue(key, out var info))
            return;

        info.count--;
        if (info.count == 0)
        {
            var handle = info.handle.Convert<T>();
            Addressables.Release(handle);
            _resDic.Remove(key);
        }
    }

    /// <summary>
    /// 批量加载资源（每个资源加载完成时回调）
    /// </summary>
    /// <typeparam name="T">资源类型</typeparam>
    /// <param name="mode">合并模式</param>
    /// <param name="onSingleLoaded">单个资源加载完成的回调</param>
    /// <param name="keys">资源地址列表</param>
    public void LoadAssetsAsync<T>(Addressables.MergeMode mode, Action<T> onSingleLoaded, params string[] keys)
    {
        if (keys == null || keys.Length == 0)
        {
            Debug.LogWarning("[AddressablesMgr] 批量加载时 keys 为空");
            return;
        }

        var key = BuildBatchKey<T>(keys);
        AsyncOperationHandle<IList<T>> handle;

        if (_resDic.TryGetValue(key, out var info))
        {
            // 已加载：增加引用计数
            info.count++;
            handle = info.handle.Convert<IList<T>>();

            if (handle.IsDone)
            {
                // 已完成则直接遍历回调
                foreach (var item in handle.Result)
                    onSingleLoaded?.Invoke(item);
            }
            else
            {
                handle.Completed += op =>
                {
                    if (op.Status == AsyncOperationStatus.Succeeded)
                    {
                        foreach (var item in op.Result)
                            onSingleLoaded?.Invoke(item);
                    }
                    else
                    {
                        Debug.LogWarning($"[AddressablesMgr] 批量资源 {key} 加载完成但状态失败");
                    }
                };
            }
            return;
        }

        // 首次加载：注意 Addressables.LoadAssetsAsync 的第二个参数是单个资源加载回调
        handle = Addressables.LoadAssetsAsync<T>(keys, onSingleLoaded, mode);
        handle.Completed += op =>
        {
            if (op.Status == AsyncOperationStatus.Failed)
            {
                Debug.LogError($"[AddressablesMgr] 批量资源 {key} 加载失败");
                if (_resDic.ContainsKey(key))
                    _resDic.Remove(key);
            }
        };

        _resDic.Add(key, new AddressablesInfo(handle));
    }

    /// <summary>
    /// 批量加载资源（统一回调，返回整个句柄）
    /// </summary>
    /// <typeparam name="T">资源类型</typeparam>
    /// <param name="mode">合并模式</param>
    /// <param name="callback">加载完成回调（整个列表）</param>
    /// <param name="keys">资源地址列表</param>
    public void LoadAssetsAsync<T>(Addressables.MergeMode mode, Action<AsyncOperationHandle<IList<T>>> callback, params string[] keys)
    {
        if (keys == null || keys.Length == 0)
        {
            Debug.LogWarning("[AddressablesMgr] 批量加载时 keys 为空");
            return;
        }

        var key = BuildBatchKey<T>(keys);
        AsyncOperationHandle<IList<T>> handle;

        if (_resDic.TryGetValue(key, out var info))
        {
            info.count++;
            handle = info.handle.Convert<IList<T>>();

            if (handle.IsDone)
            {
                callback?.Invoke(handle);
            }
            else
            {
                handle.Completed += op =>
                {
                    if (op.Status == AsyncOperationStatus.Succeeded)
                        callback?.Invoke(op);
                    else
                        Debug.LogWarning($"[AddressablesMgr] 批量资源 {key} 加载完成但状态失败");
                };
            }
            return;
        }

        // 首次加载：此重载不关心单个加载，所以第二个参数传 null
        handle = Addressables.LoadAssetsAsync<T>(keys, null, mode);
        handle.Completed += op =>
        {
            if (op.Status == AsyncOperationStatus.Succeeded)
            {
                callback?.Invoke(op);
            }
            else
            {
                Debug.LogError($"[AddressablesMgr] 批量资源 {key} 加载失败");
                if (_resDic.ContainsKey(key))
                    _resDic.Remove(key);
            }
        };

        _resDic.Add(key, new AddressablesInfo(handle));
    }

    /// <summary>
    /// 释放批量加载的资源（引用计数减1，计数为0时真正释放）
    /// </summary>
    /// <typeparam name="T">资源类型</typeparam>
    /// <param name="keys">资源地址列表</param>
    public void Release<T>(params string[] keys)
    {
        if (keys == null || keys.Length == 0)
            return;

        var key = BuildBatchKey<T>(keys);
        if (!_resDic.TryGetValue(key, out var info))
            return;

        info.count--;
        if (info.count == 0)
        {
            var handle = info.handle.Convert<IList<T>>();
            Addressables.Release(handle);
            _resDic.Remove(key);
        }
    }

    /// <summary>
    /// 清空所有已加载资源（强制释放，谨慎使用）
    /// </summary>
    public void Clear()
    {
        foreach (var kv in _resDic)
        {
            Addressables.Release(kv.Value.handle);
        }
        _resDic.Clear();

        AssetBundle.UnloadAllAssetBundles(true);
        Resources.UnloadUnusedAssets();
        GC.Collect(); // 可考虑仅在必要时调用
    }

    // ---------- 辅助方法 ----------

    /// <summary>
    /// 构建单个资源的复合 Key
    /// </summary>
    private ResourceKey BuildKey<T>(string name) => new ResourceKey(name, typeof(T));

    /// <summary>
    /// 构建批量资源的复合 Key（将多个地址和类型组合）
    /// </summary>
    private ResourceKey BuildBatchKey<T>(IEnumerable<string> keys)
    {
        // 使用排序后拼接，确保相同集合得到相同 Key（顺序无关）
        var sorted = keys.OrderBy(k => k).ToList();
        return new ResourceKey(string.Join("_", sorted), typeof(T));
    }

    // ---------- 内部类型 ----------

    /// <summary>
    /// 记录异步操作句柄和引用计数
    /// </summary>
    private class AddressablesInfo
    {
        public AsyncOperationHandle handle;
        public uint count;

        public AddressablesInfo(AsyncOperationHandle handle)
        {
            this.handle = handle;
            count = 1; // 创建时即为首次引用
        }
    }

    /// <summary>
    /// 用作字典 Key 的结构体，避免字符串拼接歧义且提升查找效率
    /// </summary>
    private readonly struct ResourceKey : IEquatable<ResourceKey>
    {
        private readonly string _name;
        private readonly Type _type;

        public ResourceKey(string name, Type type)
        {
            _name = name;
            _type = type;
        }

        public bool Equals(ResourceKey other) =>
            string.Equals(_name, other._name) && _type == other._type;

        public override bool Equals(object obj) =>
            obj is ResourceKey other && Equals(other);

        public override int GetHashCode()
        {
            unchecked
            {
                return (_name?.GetHashCode() ?? 0) * 397 ^ (_type?.GetHashCode() ?? 0);
            }
        }

        public override string ToString() => $"{_name}_{_type.Name}";
    }
}
