// iOS 打包：上线前必须逐项过的检查表
var checks = new (string item, string value, Func<string, bool> pass, string fix)[]
{
    ("Scripting Backend", "IL2CPP", v => v == "IL2CPP", "iOS 不允许 Mono，必须 IL2CPP"),
    ("Architecture", "ARM64", v => v == "ARM64", "32 位已被淘汰"),
    ("Target minimum iOS", "12.0", v => double.Parse(v) >= 12.0, "低于 12 会被新版 Xcode 拒绝"),
    ("ATS 明文 http", "已禁用", v => v == "已禁用", "Info.plist 不能保留 NSAllowsArbitraryLoads"),
    ("隐私描述 NSUserTracking", "已填写", v => v != "缺失", "缺失直接审核被拒"),
    ("Signing", "自动签名", v => v.Length > 0, "证书与描述文件必须匹配 Bundle ID"),
    ("Bitcode", "关闭", v => v == "关闭", "Xcode 14 起已废弃"),
    ("启动图 Storyboard", "已配置", v => v != "缺失", "缺失会以兼容模式上下黑边")
};

int fail = 0;
foreach (var (item, value, pass, fix) in checks)
{
    bool ok = pass(value);
    if (!ok) fail++;
    Console.WriteLine($"  [{(ok ? "通过" : "不通过")}] {item,-24} 当前: {value}");
    if (!ok) Console.WriteLine($"          修复: {fix}");
}
Console.WriteLine();
Console.WriteLine(fail == 0 ? "全部通过，可以 Archive" : $"还有 {fail} 项不达标，先修再打包");

从 Unity 到 App Store 的完整链路

Unity 只能产出 Xcode 工程，最终的签名、上传、审核都在苹果那侧完成。所以「打包」从来不是点一下 Build 就结束。

  1. Player Settings：Bundle Identifier、版本号、图标、启动图、方向、最低系统版本。
  2. Build Settings：Scripting Backend 必须 IL2CPP，Architecture 选 ARM64。
  3. 导出 Xcode 工程 → 用 Xcode 打开，配置 Team 与 Signing。
  4. Product → Archive → Distribute App → App Store Connect 上传。
  5. 在 App Store Connect 填元数据、截图、隐私清单，提交审核。

八个最容易卡住的设置

项目要求不做的后果
Scripting BackendIL2CPPiOS 不支持 Mono，直接构建失败
ArchitectureARM6432 位早已被拒
ATS（App Transport Security）禁止明文 http所有 http 请求静默失败
隐私描述文案用到的权限都要写用途审核直接打回
IDFA / 广告标识不用就别声明声明了却不用也会被拒
启动图 Storyboard必须配置以兼容模式运行，上下黑边
Bitcode关闭（已废弃）上传报错
第三方 SDK隐私清单 PrivacyInfo.xcprivacy2024 起强制要求

真机上才出现的问题

  • 发热掉帧：编辑器的性能数据完全不可参考，必须在真机用 Xcode Instruments 看。
  • 内存被系统杀掉：iOS 没有 swap，纹理超限直接 SIGKILL，表现为「闪退但没日志」。
  • 首次启动白屏：首包资源解压 + 热更检查都在启动阶段，用启动图遮住并给进度。
  • 支付/登录回调：URL Scheme 与 Universal Link 要在 Info.plist 与后台同时配置。
