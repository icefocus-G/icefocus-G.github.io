// 程序框架：服务定位 + 流程状态机
using System;
using System.Collections.Generic;

public static class Locator
{
    private static readonly Dictionary<Type, object> map = new();
    public static void Reg<T>(T svc) where T : class => map[typeof(T)] = svc;
    public static T Get<T>() where T : class => map.TryGetValue(typeof(T), out var o) ? (T)o : null;
}

public class AudioSvc { public void Play(string s) => Console.WriteLine($"  [Audio] {s}"); }
public class SaveSvc { public void Save() => Console.WriteLine("  [Save] 已存档"); }

public static class Program
{
    public static void Main()
    {
        Locator.Reg(new AudioSvc());
        Locator.Reg(new SaveSvc());

        Console.WriteLine("服务定位：");
        Locator.Get<AudioSvc>()?.Play("bgm_main");
        Locator.Get<SaveSvc>()?.Save();
        Console.WriteLine($"取未注册的服务：{(Locator.Get<object>() == null ? "null（不抛异常，但要自己判空）" : "有值")}");

        Console.WriteLine();
        Console.WriteLine("流程状态机（登录 -> 选角 -> 战斗 -> 结算）：");
        foreach (var f in new[] { "Login", "SelectRole", "Battle", "Result" })
        {
            Console.WriteLine($"  进入 {f} ...");
            if (f == "Battle") Console.WriteLine("    加载战斗场景、注册战斗服务");
            if (f == "Result") Console.WriteLine("    战斗服务反注册、回主城");
        }
        Console.WriteLine("流程结束");
    }
}
