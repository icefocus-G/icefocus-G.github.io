// Android 打包：签名与目标 API 校验
var cfg = new Dictionary<string, string>
{
    ["Scripting Backend"] = "IL2CPP",
    ["Target Architectures"] = "ARM64",
    ["Target API Level"] = "34",
    ["Minimum API Level"] = "23",
    ["Keystore"] = "release.keystore",
    ["Keystore Pass"] = "已设置",
    ["Build App Bundle"] = "开启",
    ["Internet Permission"] = "有"
};

bool Ok(string k, Func<string, bool> f) => cfg.TryGetValue(k, out var v) && f(v);

var rules = new (string name, bool ok, string why)[]
{
    ("IL2CPP（64 位必需）", Ok("Scripting Backend", v => v == "IL2CPP"), "Mono 只能出 32 位，Google Play 已拒绝"),
    ("ARM64 架构", Ok("Target Architectures", v => v.Contains("ARM64")), "缺 ARM64 无法上架"),
    ("Target API ≥ 34", Ok("Target API Level", v => int.Parse(v) >= 34), "每年上架门槛会抬，必须跟上"),
    ("Release 签名", Ok("Keystore Pass", v => v == "已设置"), "用 debug 签名无法上架，且无法覆盖升级"),
    ("AAB 格式", Ok("Build App Bundle", v => v == "开启"), "Play 商店要求 AAB，不再收 APK"),
    ("网络权限", Ok("Internet Permission", v => v == "有"), "漏了就是「游戏连不上服务器」")
};

int bad = 0;
foreach (var (name, ok, why) in rules)
{
    if (!ok) bad++;
    Console.WriteLine($"  [{(ok ? "通过" : "不通过")}] {name,-22} {(ok ? "" : why)}");
}
Console.WriteLine();
Console.WriteLine($"未通过 {bad} 项");
Console.WriteLine();
Console.WriteLine("签名要点：keystore 一旦丢失，就无法给已上架的应用发更新包——务必备份两处。");

Android 打包链路

Android 的坑集中在「签名」和「目标 API」两件事上：签名决定能不能上架与覆盖升级，目标 API 决定能不能通过商店的年度审核。

  1. Player Settings：Package Name（反写域名，上架后不可改）、版本号、图标、方向。
  2. Scripting Backend 选 IL2CPP，Target Architectures 勾 ARM64（Google Play 强制）。
  3. 创建 keystore（Build → Keystore Manager），密码与别名记牢并备份。
  4. Player Settings → Publishing Settings 指定 keystore、别名、密码。
  5. Build 出 AAB（上架用）或 APK（自测 / 第三方渠道用）。

上线检查表

项目要求说明
Target API Level跟当年要求（34+）每年抬高，不达标无法更新
Minimum API Level23 左右太低要兼容大量老设备，太高丢用户
ArchitecturesARM64 必选只出 ARMv7 会被拒
签名Release keystoredebug 签名无法上架，也无法覆盖升级
格式AABPlay 商店已不收 APK
权限按需声明多余权限会被要求说明用途
数据安全表单如实填写与实际采集不符会被下架

真机适配高频问题

  • keystore 丢失：等于永远无法给已上架应用发更新，必须两处备份（仓库外 + 密码管理）。
  • 刘海屏 / 挖孔屏：UI 要放 Safe Area 内，尤其是顶部状态栏与底部手势区。
  • 返回键：Android 物理返回键要映射到「关闭当前弹窗 → 返回上一级 → 二次确认退出」，别直接杀掉进程。
  • 后台被杀：切后台超过几分钟进程可能被回收，回来必须能恢复到正确界面。
  • 多渠道：不同渠道包用同一 keystore，靠渠道号区分，不要把逻辑写死在构建里。
