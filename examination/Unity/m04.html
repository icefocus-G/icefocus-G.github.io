// 网络：长度前缀协议解决粘包 / 拆包
var stream = new List<byte>();
void Send(string msg)
{
    var body = System.Text.Encoding.UTF8.GetBytes(msg);
    var head = BitConverter.GetBytes(body.Length);   // 4 字节长度前缀
    stream.AddRange(head); stream.AddRange(body);
}

Send("login:Tom");
Send("move:12,34");
Send("chat:你好");   // 三次发送在 TCP 里可能粘在一起

Console.WriteLine($"缓冲区共 {stream.Count} 字节，模拟一次收到全部数据：");
var buf = stream.ToArray();
int off = 0, packets = 0;
while (off + 4 <= buf.Length)                     // 先读长度
{
    int len = BitConverter.ToInt32(buf, off);
    if (off + 4 + len > buf.Length) { Console.WriteLine($"  剩余不足 {len} 字节，等下次数据（拆包）"); break; }
    var msg = System.Text.Encoding.UTF8.GetString(buf, off + 4, len);
    packets++;
    Console.WriteLine($"  包 {packets}: {msg}  ({len} 字节)");
    off += 4 + len;
}
Console.WriteLine($"解析出 {packets} 个完整包，剩余 {buf.Length - off} 字节待续");
