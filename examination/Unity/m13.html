// Newtonsoft.Json：嵌套对象、字典、自定义设置
using System;
using System.Collections.Generic;
using Newtonsoft.Json;
using Newtonsoft.Json.Converters;

public class Attr { public int Hp { get; set; } public int Atk { get; set; } }

public class Player
{
    public int Id { get; set; }
    public string Name { get; set; }
    public Attr Attr { get; set; }
    public Dictionary<string, int> Items { get; set; } = new();
    public DateTime Created { get; set; }
}

public static class Program
{
    public static void Main()
    {
        var p = new Player
        {
            Id = 1001,
            Name = "唐老狮",
            Attr = new Attr { Hp = 100, Atk = 15 },
            Items = new() { ["木剑"] = 1, ["药水"] = 5 },
            Created = new DateTime(2026, 7, 10, 16, 7, 13)
        };

        var setting = new JsonSerializerSettings
        {
            Formatting = Formatting.Indented,
            NullValueHandling = NullValueHandling.Ignore,
            Converters = { new IsoDateTimeConverter { DateTimeFormat = "yyyy-MM-dd HH:mm:ss" } }
        };

        var json = JsonConvert.SerializeObject(p, setting);
        Console.WriteLine("序列化：");
        Console.WriteLine(json);

        var back = JsonConvert.DeserializeObject<Player>(json);
        Console.WriteLine();
        Console.WriteLine($"反序列化：Id={back.Id} Name={back.Name} Hp={back.Attr.Hp} 道具数={back.Items.Count} 创建={back.Created:yyyy-MM-dd HH:mm:ss}");

        Console.WriteLine();
        Console.WriteLine("JsonUtility 做不到、Newtonsoft 可以的三件事：");
        Console.WriteLine("  字典 Dictionary<string,int>  -> " + (json.Contains("\"木剑\"") ? "OK" : "丢失"));
        Console.WriteLine("  null 值控制 NullValueHandling -> OK");
        Console.WriteLine("  属性 Property（JsonUtility 只认 public 字段）-> OK");
    }
}

3 : JsonUtlity 和 LitJson 对比 知识点

C#
知识点一 JsonUtlity和LitJson相同点
1.他们都是用于Json的序列化反序列化
2.Json文档编码格式必须是UTF-8
3.都是通过静态类进行方法调用


知识点二 JsonUtlity和LitJson不同点
1.JsonUtlity是Unity自带，LitJson是第三方需要引用命名空间
2.JsonUtlity使用时自定义类需要加特性,LitJson不需要
3.JsonUtlity支持私有变量(加特性),LitJson不支持
4.JsonUtlity不支持字典,LitJson支持(但是键只能是字符串)
5.JsonUtlity不能直接将数据反序列化为数据集合(数组字典),LitJson可以
6.JsonUtlity对自定义类不要求有无参构造，LitJson需要
7.JsonUtlity存储空对象时会存储默认值而不是null，LitJson会存null


知识点三 如何选择两者
根据实际需求
建议使用LitJson
原因：LitJson不用加特性，支持字典，支持直接反序列化为数据集合，存储null更准确

4 : 实现 JsonUtlity 和 Litjson 动态选择的管理类 进行 json 的存储和读取

C#
using LitJson;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

/// <summary>
/// 序列化和反序列化Json时  使用的是哪种方案
/// </summary>
public enum JsonType
{
    JsonUtlity,
    LitJson,
}

/// <summary>
/// Json数据管理类 主要用于进行 Json的序列化存储到硬盘 和 反序列化从硬盘中读取到内存中
/// </summary>
public class JsonMgr
{
    private static JsonMgr instance = new JsonMgr();
    public static JsonMgr Instance => instance;

    private JsonMgr() { }

    //存储Json数据 序列化
    public void SaveData(object data, string fileName, JsonType type = JsonType.LitJson)
    {
        //确定存储路径
        string path = Application.persistentDataPath + "/" + fileName + ".json";
        //序列化 得到Json字符串
        string jsonStr = "";
        switch (type)
        {
            case JsonType.JsonUtlity:
                jsonStr = JsonUtility.ToJson(data);
                break;
            case JsonType.LitJson:
                jsonStr = JsonMapper.ToJson(data);
                break;
        }
        //把序列化的Json字符串 存储到指定路径的文件中
        File.WriteAllText(path, jsonStr);
    }

    //读取指定文件中的 Json数据 反序列化
    public T LoadData<T>(string fileName, JsonType type = JsonType.LitJson) where T : new()
    {
        //确定从哪个路径读取
        //首先先判断 默认数据文件夹中是否有我们想要的数据 如果有 就从中获取
        string path = Application.streamingAssetsPath + "/" + fileName + ".json";
        //先判断 是否存在这个文件
        //如果不存在默认文件 就从 读写文件夹中去寻找
        if(!File.Exists(path))
            path = Application.persistentDataPath + "/" + fileName + ".json";
        //如果读写文件夹中都还没有 那就返回一个默认对象
        if (!File.Exists(path))
            return new T();

        //进行反序列化
        string jsonStr = File.ReadAllText(path);
        //数据对象
        T data = default(T);
        switch (type)
        {
            case JsonType.JsonUtlity:
                data = JsonUtility.FromJson<T>(jsonStr);
                break;
            case JsonType.LitJson:
                data = JsonMapper.ToObject<T>(jsonStr);
                break;
        }

        //把对象返回出去
        return data;
    }
}

Newtonsoft.Json 使用教程

UnitNewtonsoft.Json（也称为 Json.NET）是 .NET 平台最流行的 JSON 序列化/反序列化库，在游戏开发中常用于存档、配置读取、网络通信等场景。下面我将介绍其核心 API，并设计一个可直接使用的管理类。

一、Newtonsoft.Json 核心 API 概览

API说明
JsonConvert.SerializeObject(object value)将对象序列化为 JSON 字符串
JsonConvert.DeserializeObject<T>(string json)将 JSON 字符串反序列化为指定类型的对象
JObject / JArray / JToken表示任意 JSON 结构，用于动态解析或构建
JsonSerializerSettings配置序列化/反序列化的行为（如忽略空值、日期格式、循环引用处理等）
JsonConverter自定义转换器，用于处理特殊类型或自定义序列化逻辑

二、设计一个 JsonManager 管理类

为了方便在游戏项目中使用，我们可以封装一个静态类 JsonManager，提供常用的序列化/反序列化方法，并支持自定义设置。

csharp

PLAIN
using System;
using System.IO;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

/// <summary>
/// JSON 管理器：封装 Newtonsoft.Json 的常用操作，
/// 提供统一的序列化/反序列化接口，方便游戏逻辑直接调用。
/// </summary>
public static class JsonManager
{
    // 默认序列化设置：适合大多数游戏场景
    private static readonly JsonSerializerSettings DefaultSettings = new JsonSerializerSettings
    {
        // 忽略 null 值，减小存档体积
        NullValueHandling = NullValueHandling.Ignore,
        
        // 忽略 JSON 中不存在的属性，防止反序列化时出错
        MissingMemberHandling = MissingMemberHandling.Ignore,
        
        // 处理循环引用（如父子对象互相引用）
        ReferenceLoopHandling = ReferenceLoopHandling.Ignore,
        
        // 使用缩进格式，便于阅读（调试时可开启）
        Formatting = Formatting.Indented,
        
        // 日期格式统一为 ISO 8601
        DateFormatString = "yyyy-MM-ddTHH:mm:ssZ"
    };

    /// <summary>
    /// 将对象序列化为 JSON 字符串。
    /// </summary>
    /// <typeparam name="T">对象类型</typeparam>
    /// <param name="obj">要序列化的对象</param>
    /// <returns>JSON 字符串</returns>
    public static string ToJson<T>(T obj)
    {
        return JsonConvert.SerializeObject(obj, DefaultSettings);
    }

    /// <summary>
    /// 将对象序列化为紧凑格式的 JSON 字符串（无缩进，忽略 null）。
    /// </summary>
    public static string ToJsonCompact<T>(T obj)
    {
        var settings = new JsonSerializerSettings
        {
            NullValueHandling = NullValueHandling.Ignore,
            Formatting = Formatting.None
        };
        return JsonConvert.SerializeObject(obj, settings);
    }

    /// <summary>
    /// 将 JSON 字符串反序列化为指定类型的对象。
    /// </summary>
    /// <typeparam name="T">目标类型</typeparam>
    /// <param name="json">JSON 字符串</param>
    /// <returns>反序列化得到的对象，若失败则返回默认值</returns>
    public static T FromJson<T>(string json)
    {
        try
        {
            return JsonConvert.DeserializeObject<T>(json, DefaultSettings);
        }
        catch (JsonException ex)
        {
            // 在实际项目中可记录日志或抛出异常
            Console.WriteLine($"JSON 反序列化失败: {ex.Message}");
            return default(T);
        }
    }

    /// <summary>
    /// 将对象序列化并保存到文件。
    /// </summary>
    /// <typeparam name="T">对象类型</typeparam>
    /// <param name="path">文件路径</param>
    /// <param name="obj">要保存的对象</param>
    public static void SaveToFile<T>(string path, T obj)
    {
        string json = ToJson(obj);
        File.WriteAllText(path, json);
    }

    /// <summary>
    /// 从文件读取 JSON 并反序列化为对象。
    /// </summary>
    /// <typeparam name="T">目标类型</typeparam>
    /// <param name="path">文件路径</param>
    /// <returns>反序列化得到的对象</returns>
    public static T LoadFromFile<T>(string path)
    {
        if (!File.Exists(path))
            return default(T);

        string json = File.ReadAllText(path);
        return FromJson<T>(json);
    }

    /// <summary>
    /// 解析 JSON 字符串为 JObject，用于动态读取字段（无需定义强类型类）。
    /// </summary>
    /// <param name="json">JSON 字符串</param>
    /// <returns>JObject 实例</returns>
    public static JObject Parse(string json)
    {
        return JObject.Parse(json);
    }

    /// <summary>
    /// 将任意对象转换为 JToken（动态 JSON 结构）。
    /// </summary>
    public static JToken ToJToken(object obj)
    {
        return JToken.FromObject(obj);
    }

    /// <summary>
    /// 将 JToken 转换为指定类型的对象。
    /// </summary>
    public static T FromJToken<T>(JToken token)
    {
        return token.ToObject<T>();
    }
}

四、注意事项

  • 引用库：在 Unity 中，需通过 Package Manager 添加 com.unity.nuget.newtonsoft-json 包，或从 Asset Store 导入 Json.NET。在普通 .NET 项目中通过 NuGet 安装 Newtonsoft.Json。
  • 性能：频繁序列化/反序列化大对象时，可考虑缓存 JsonSerializer 实例或使用 JsonConvert 的预编译代理，但对于大多数游戏场景，直接调用即可。
  • 类型安全：反序列化时若 JSON 结构与目标类不完全匹配，默认会忽略缺失成员，但多余字段可能导致异常（可通过 MissingMemberHandling.Ignore 避免）。
  • 特殊类型：Unity 中的 Vector3、Quaternion 等结构默认无法直接序列化，需要自定义 JsonConverter。管理类中未包含，可根据需要扩展。

通过这个 JsonManager，你可以在游戏项目中快速、统一地处理 JSON 数据，减少重复代码。
