// ScriptableObject：配置资产按 ID 索引 + 覆盖合并
using System;
using System.Linq;

public record ItemCfg(int Id, string Name, int Atk, int Hp);

public static class Program
{
    public static void Main()
    {
        var baseCfg = new[]
        {
            new ItemCfg(1001, "木剑", 5, 0),
            new ItemCfg(1002, "铁剑", 12, 0),
            new ItemCfg(2001, "皮甲", 0, 20)
        };

        // 活动版本只改两条，不动原表
        var patch = new[]
        {
            new ItemCfg(1002, "铁剑+", 18, 5),
            new ItemCfg(2002, "链甲", 0, 45)
        };

        var index = baseCfg.ToDictionary(x => x.Id);
        foreach (var p in patch) index[p.Id] = p;   // 同 ID 覆盖，新 ID 追加

        Console.WriteLine("合并后的配置表（按 ID 排序）：");
        foreach (var kv in index.OrderBy(k => k.Key))
        {
            var c = kv.Value;
            var origin = baseCfg.FirstOrDefault(b => b.Id == c.Id);
            string tag = origin == null ? "新增" : (origin == c ? "" : "覆盖");
            Console.WriteLine($"  {c.Id}  {c.Name,-4} 攻{c.Atk,3} 血{c.Hp,3}  {tag}");
        }

        Console.WriteLine();
        Console.WriteLine("为什么用 SO 而不是直接读 Excel：");
        Console.WriteLine("  1) SO 是 Unity 序列化资产，引用关系（贴图/预制体）不会断");
        Console.WriteLine("  2) 改动走版本管理，diff 清晰");
        Console.WriteLine("  3) 运行时零解析开销，直接反序列化成对象");
    }
}

第1章：ScriptableObject概述

01：ScriptableObject概述

Unity 提供的一个数据存储基类 ,文件配置、数据复用、更好的处理数据带来的多态行为

如果你多个物体引用同一个 ScriptableObject 资源： 所有物体拿到的是同一个 SO 实例，同一个内存地址。改 SO 里面的数据，所有物体读取全部一起变。

场景里多个 GameObject，每个都挂 Lesson.cs(MonoBehaviour)：

✅ 每一个对象，都是独立脚本实例，内存地址互不相同，就会加大内存开销

ScriptableObject 就是自定义的数据资源

ScriptableObject 的适用范围极广，本质上任何需要结构化、可编辑、可复用、可持久化的数据都可以用它来承载。它不仅用于游戏逻辑数据，也用于编辑器工具、事件通信、资源管理等，是 Unity 数据驱动架构的核心组件

数据复用，编辑模式下具有数据持久化，运行模式下 ScriptableObject 不具备持久化特性（修改数据对象，并不会保存在本地）

PLAIN
ScriptableObject的好处
1.我们可以直接在Inspector窗口编辑配置数据，可以利用它来做配置文件
2.处理重复数据，减少数据拷贝时造成的内存占用，可以利用它来做公共数据
3.可以更方便的处理数据带来的多态行为

指的是不管在编辑器模式还是在发布后都 不会持久化的数据

只存在于内存中，不占硬盘空间，就减少了包体的大小，游戏结束后就内存释放掉，不会存在本地

public class RandomPlayAudio : AuidoPlayBase

    //希望随机播放的音效文件
    public List<AudioClip> clips;

    //重写基类的Play播放方法
    public override void Play(AudioSource source)
    {
        // 如果音效列表为空，直接退出，避免报错
        if (clips.Count == 0)
            return;

        // 执行播放
        source.Play();
    }
}

02 单列模式的数据文件基类

实现 通过生成的 数据资源的名字 点出的方式获取 数据资源里的信息，不用再去拖入

C#
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class SingleScriptableObject<T> :ScriptableObject where T:ScriptableObject
{
    private static T instance;

    public static T Instance
    {
        get
        {
            //如果为空 首先应该去资源路劲下加载 对应的 数据资源文件
            if (instance == null)
            {
                //我们定两个规则
                //1.所有的 数据资源文件都放在 Resources文件夹下的ScriptableObject中
                //2.需要复用的 唯一的数据资源文件名 我们定一个规则：和类名是一样的
                instance = Resources.Load<T>("ScriptableObject/" + typeof(T).Name);
            }
            //如果没有这个文件 为了安全起见 我们可以在这直接创建一个数据
            if(instance==null)
            {
                instance = CreateInstance<T>();
            }
            //甚至可以在这里 从json当中读取数据，但是我不建议用ScriptableObject来做数据持久化

            return instance;
        }
    }
}
