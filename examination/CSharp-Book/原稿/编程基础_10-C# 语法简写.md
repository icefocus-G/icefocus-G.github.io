```csharp
using UnityEngine;

class Example<T> : MonoBehaviour where T : class
{
    // 私有静态存储容器，所有写法共用
    private static T instance;

    // ========== 写法1：老式完整只读属性（全版本兼容，新手好理解）==========
    public static T Instance
    {
        get
        {
            return instance;
        }
    }

     ========== 写法2：表达式体单行只读（你UI单例推荐用，最简只读）==========
     只有get、无set才能这么写，外部不能赋值修改
     public static T Instance => instance;

     ========== 写法3：老式完整 可读可写（带set，外部能修改单例，项目不推荐）==========

    public static T Instance
    {
        get { return instance; }
        set { instance = value; }
    }

      ========== 写法4：get/set分别用表达式简写（带set场景最简）==========

    public static T Instance
    {
        get => instance;
        set => instance = value;
    }

}
```
