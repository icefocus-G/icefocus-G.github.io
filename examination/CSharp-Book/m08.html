        static void Calc(uint num)
        {
            Console.Write("{0}的二进制是：", num);
            Stack stack = new Stack();
            while(true)
            {
                stack.Push(num % 2);
                num /= 2;
                if (num == 1)
                {
                    stack.Push(num);
                    break;
                }
            }
            //循环弹栈
            while(stack.Count > 0)
            {
                Console.Write(stack.Pop());
            }
